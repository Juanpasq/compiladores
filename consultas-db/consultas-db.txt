Obtener la distribución de Sector/Nomenclador por Categoría.

db.empleados_talles.aggregate([

  //contamos cuántas personas hay por cada combinación exacta

    $group: {

        categoria: "$categoria",

      },
      cantidad: { $sum: 1 }
    }
  },
  // re-agrupamos  por categoría, y armamos la distribución en una lista

    $group: {
      _id: "$_id.categoria", // El grupo principal ahora es la Categoría

        $push: {
          nomenclador: "$_id.nomenclador",
          cantidad: "$cantidad"
        }
      },
      total_en_categoria: { $sum: "$cantidad" } // ¡Un dato extra que le suma al TP!
    }
  },
  
  {
    $sort: { "_id": 1 }
  }
])

Obtener la distribución de tallas por prenda/accesorios sobre:
 
- Personal activo/inactivo.

db.empleados_talles.aggregate([
  // Transformar el objeto de talles en un Array que Mongo pueda leer
  // esto convierte { pantalon: "48" } en [{ k: "pantalon", v: "48" }]
  {
    $project: {
      estado: 1, // Mantenemos el campo Estado
      talles_array: { $objectToArray: "$talles" } 
    }
  },
  
  // separar ese array para que cada prenda tenga su propio registro temporal
  {
    $unwind: "$talles_array"
  },
  
  // agrupamos y contamos cada combinacion exacta (estado + prenda + talle)
  {
    $group: {
      _id: {
        estado: "$estado",
        prenda: "$talles_array.k", // "k" es la key (ej: pantalon)
        talle: "$talles_array.v"   // "v" es el value (ej: "48")
      },
      cantidad: { $sum: 1 }
    }
  },
  
  // ordenamos para la distribucion
  {
    $group: {
      _id: {
        estado: "$_id.estado",
        prenda: "$_id.prenda"
      },
      distribucion_talles: {
        $push: {
          talle: "$_id.talle",
          cantidad: "$cantidad"
        }
      },
      total_unidades: { $sum: "$cantidad" }
    }
  },
  
  // alfabeticamente
  {
    $sort: { "_id.estado": 1, "_id.prenda": 1 }
  }
])

Por Prenda, obtener la cantidad total a solicitar y distribuir y a qué Sector aplican (+4% stock)

db.sectores_prendas.aggregate([
  // 1. y 2. Desarmamos las listas
  { 
    $unwind: "$sectores" 
  },
  { 
    $unwind: "$sectores.kit" 
  },
  
  // 3. Agrupamos por Prenda y sumamos la base
  {
    $group: {
      _id: "$sectores.kit.prenda",
      cantidad_base: { $sum: "$sectores.kit.cantidad" },
      sectores_que_aplican: { $addToSet: "$sectores.descripcion" }
    }
  },
  
  // 4.Calculamos el 4% extra y el Total Final
  {
    $addFields: {
      adicional_stock_4_porciento: { 
        $ceil: { $multiply: [ "$cantidad_base", 0.04 ] } 
      },
      total_final_a_comprar: { 
        $ceil: { $multiply: [ "$cantidad_base", 1.04 ] } 
      }
    }
  },
  
  // 5. Ordenamos
  {
    $sort: { "_id": 1 }
  }
])

db.empleados_talles.aggregate([
  // 1. Aplanamos los talles convertimos los campos talle_remera, talle_pantalon, etc. en una lista
  {
    $project: {
      frente: 1,
      sector: 1, // Cambialo por "categoria" si ese es el nombre en tu colección de empleados
      talles_unificados: [
        { prenda: "Remera/Chomba", talle: "$talle_remera" },
        { prenda: "Pantalon", talle: "$talle_pantalon" },
        { prenda: "Calzado", talle: "$talle_calzado" }
      ]
    }
  },

  // 2. Desarmamos  lista para tener un documento por cada prenda de cada empleado
  { $unwind: "$talles_unificados" },

  // 3. Agrupamos por Frente, Sector, Prenda y el valor del Talle
  {
    $group: {
      _id: {
        frente: "$frente",
        sector: "$sector",
        prenda: "$talles_unificados.prenda",
        talle: "$talles_unificados.talle"
      },
      cantidad_personas: { $sum: 1 }
    }
  },

  // 4. Aplicamos criterio del 4% adicional sobre la demanda  de talles
  {
    $addFields: {
      stock_seguridad_4_porciento: { 
        $ceil: { $multiply: ["$cantidad_personas", 0.04] } 
      },
      total_a_solicitar: { 
        $ceil: { $multiply: ["$cantidad_personas", 1.04] } 
      }
    }
  },

  // 5. Ordenamos 
  {
    $sort: { 
      "_id.frente": 1, 
      "_id.sector": 1, 
      "_id.prenda": 1, 
      "_id.talle": 1 
    }
  }
])
